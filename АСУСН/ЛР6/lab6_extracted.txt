11: 2. Теоретическая часть
12: 2.1. Что такое «виртуальный ПЛК» на Python
13: Реальный ПЛК – это промышленный компьютер, который циклически выполняет программу управления. Мы можем смоделировать его поведение на Python:
14: Цикл ПЛК – бесконечный цикл с фиксированным периодом (например, 100 мс). На каждом шаге:
15: Читаются входные сигналы (от датчиков/имитатора);
16: Выполняется программа управления (ПИД, логика);
17: Записываются выходные сигналы (в Modbus-регистры или напрямую имитатору).
18: Modbus TCP Server – виртуальный ПЛК «выставляет» свои регистры в сеть, чтобы SCADA и другие ПЛК могли их читать.
19: Modbus TCP Client – виртуальный ПЛК может читать данные других ПЛК и записывать команды.
20: Таким образом, каждый «ПЛК» – это Python-скрипт, который:
21: запускает Modbus TCP Server (через pymodbus);
22: выполняет циклическую программу управления;
23: обменивается данными с имитатором объекта (через сокет или файл).

67: 4.2. Реализация имитатора объекта 
68: Создать файл plant_simulator.py:
69: python
70: import socket
71: import json
72: import time
74: class PlantSimulator:
75:     """
76:     Имитатор технологического объекта:
77:     резервуар с жидкостью, нагреватель, насос, клапан.
78:     """
79:     def __init__(self):
80:         self.temperature = 20.0
81:         self.level = 50.0
82:         self.pressure = 1.0
83:         self.env_temp = 20.0
85:         self.K_heat = 0.5
86:         self.K_cool = 0.05
87:         self.K_in = 2.0
88:         self.K_out = 1.0
90:         # Управляющие воздействия (от ПЛК №2)
91:         self.heater_power = 0.0
92:         self.pump_on = False
93:         self.valve_open = False
95:     def step(self, dt=0.1):
96:         """Один шаг моделирования."""
97:         # Нагрев/охлаждение
98:         dT = (self.heater_power * self.K_heat
99:               - self.K_cool * (self.temperature - self.env_temp)) * dt
100:         self.temperature += dT
102:         # Уровень
103:         q_in = self.K_in if self.pump_on else 0.0
104:         q_out = self.K_out if self.valve_open else 0.0
105:         self.level += (q_in - q_out) * dt
107:         # Ограничения
108:         self.level = max(0.0, min(100.0, self.level))
109:         self.temperature = max(0.0, min(200.0, self.temperature))
111:         # Давление
112:         self.pressure = 0.5 + 0.05 * self.level + 0.02 * self.temperature
114:         return {
115:             'temperature': round(self.temperature, 2),
116:             'level': round(self.level, 2),
117:             'pressure': round(self.pressure, 3)
118:         }
121: def run_server(host='127.0.0.1', port=9999):
122:     plant = PlantSimulator()
123:     server = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
124:     server.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
125:     server.bind((host, port))
126:     server.listen(1)
127:     print(f"Имитатор объекта запущен на {host}:{port}")
129:     conn, addr = server.accept()
130:     print(f"Подключение от {addr}")
132:     try:
133:         while True:
134:             # Приём команд от ПЛК №2
135:             data = conn.recv(1024)
136:             if not data:
137:                 break
138:             commands = json.loads(data.decode('utf-8'))
139:             plant.heater_power = commands.get('heater_power', 0.0)
140:             plant.pump_on = commands.get('pump_on', False)
141:             plant.valve_open = commands.get('valve_open', False)
143:             # Шаг моделирования
144:             state = plant.step(dt=0.1)
146:             # Отправка состояния
147:             conn.sendall(json.dumps(state).encode('utf-8'))
148:             time.sleep(0.1)
149:     finally:
150:         conn.close()
151:         server.close()
154: if __name__ == '__main__':
155:     run_server()
156: Пояснение: имитатор – это TCP-сервер. ПЛК №2 подключается к нему как клиент, отправляет команды (мощность нагрева, насос, клапан) и получает текущее состояние объекта.
157: 4.3. Реализация ПЛК №1 
158: Создать файл plc1.py:
159: python
160: from pymodbus.server import StartTcpServer
161: from pymodbus.datastore import (
162:     ModbusSequentialDataBlock,
163:     ModbusSlaveContext,
164:     ModbusServerContext
165: )
166: import threading
167: import time
169: # Хранилище данных ПЛК №1
170: # HR[0] = Temperature × 10
171: # HR[1] = Heater_Power × 10
172: # HR[2] = Setpoint × 10
173: # HR[3] = Level
174: # HR[4] = Pressure × 100
175: # CO[0] = Pump_On
176: # CO[1] = Valve_Open
177: store = ModbusSlaveContext(
178:     di=ModbusSequentialDataBlock(0, [0]*100),
179:     co=ModbusSequentialDataBlock(0, [0]*100),
180:     hr=ModbusSequentialDataBlock(0, [0]*100),
181:     ir=ModbusSequentialDataBlock(0, [0]*100)
182: )
183: context = ModbusServerContext(slaves=store, single=True)
185: # Состояние ПЛК
186: state = {
187:     'temperature': 20.0,
188:     'setpoint': 60.0,
189:     'heater_power': 0.0,
190:     'integral': 0.0,
191:     'prev_error': 0.0,
192:     'Kp': 2.0, 'Ki': 0.5, 'Kd': 0.1
193: }
196: def pid_step():
197:     """Один шаг ПИД-регулятора."""
198:     error = state['setpoint'] - state['temperature']
199:     state['integral'] += error * 0.1
200:     derivative = (error - state['prev_error']) / 0.1
201:     output = (state['Kp'] * error
202:               + state['Ki'] * state['integral']
203:               + state['Kd'] * derivative)
204:     state['prev_error'] = error
206:     # Ограничение
207:     output = max(0.0, min(100.0, output))
208:     state['heater_power'] = output
209:     return output
212: def control_loop():
213:     """Цикл управления ПЛК №1 (100 мс)."""
214:     while True:
215:         # Чтение данных от ПЛК №2 через Modbus (или из локального состояния)
216:         # В упрощённом варианте – температура обновляется через Modbus-клиент
217:         # Здесь для простоты: температура берётся из глобального состояния,
218:         # которое обновляет отдельный клиент (см. plc1_modbus_client.py)
220:         heater = pid_step()
222:         # Запись в Modbus-регистры
223:         store.setValues(3, 0, [int(state['temperature'] * 10)])
224:         store.setValues(3, 1, [int(heater * 10)])
225:         store.setValues(3, 2, [int(state['setpoint'] * 10)])
227:         time.sleep(0.1)
230: def modbus_server():
231:     """Запуск Modbus TCP сервера ПЛК №1."""
232:     StartTcpServer(context, address=("127.0.0.1", 5020))
235: if __name__ == '__main__':
236:     # Запуск сервера в отдельном потоке
237:     t_server = threading.Thread(target=modbus_server, daemon=True)
238:     t_server.start()
240:     print("ПЛК №1 запущен. Modbus TCP на порту 5020.")
241:     print("HR[0]=T, HR[1]=Heater, HR[2]=Setpoint")
243:     # Запуск цикла управления
244:     control_loop()
245: Пояснение: ПЛК №1 реализует ПИД-регулятор и выставляет свои регистры в Modbus TCP на порту 5020. SCADA и ПЛК №2 могут их читать.
246: 4.4. Реализация ПЛК №2 
247: Создать файл plc2.py:
248: python
249: from pymodbus.server import StartTcpServer
250: from pymodbus.datastore import (
251:     ModbusSequentialDataBlock,
252:     ModbusSlaveContext,
253:     ModbusServerContext
254: )
255: from pymodbus.client import ModbusTcpClient
256: import socket
257: import json
258: import threading
259: import time
261: # Хранилище данных ПЛК №2
262: store = ModbusSlaveContext(
263:     di=ModbusSequentialDataBlock(0, [0]*100),
264:     co=ModbusSequentialDataBlock(0, [0]*100),
265:     hr=ModbusSequentialDataBlock(0, [0]*100),
266:     ir=ModbusSequentialDataBlock(0, [0]*100)
267: )
268: context = ModbusServerContext(slaves=store, single=True)
270: # Состояние ПЛК №2
271: state = {
272:     'temperature': 20.0,
273:     'level': 50.0,
274:     'pressure': 1.0,
275:     'pump_on': False,
276:     'valve_open': False,
277:     'heater_power': 0.0
278: }
280: # Соединение с имитатором объекта
281: plant_socket = None
284: def connect_to_plant():
285:     """Подключение к имитатору объекта."""
286:     global plant_socket
287:     plant_socket = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
288:     plant_socket.connect(('127.0.0.1', 9999))
289:     print("ПЛК №2 подключён к имитатору объекта")
292: def read_from_plc1():
293:     """Чтение данных от ПЛК №1 через Modbus TCP."""
294:     client = ModbusTcpClient('127.0.0.1', port=5020)
295:     client.connect()
296:     try:
297:         hr = client.read_holding_registers(0, 3, slave=1)
298:         if not hr.isError():
299:             state['temperature'] = hr.registers[0] / 10.0
300:             state['heater_power'] = hr.registers[1] / 10.0
301:             setpoint = hr.registers[2] / 10.0
302:     finally:
303:         client.close()
306: def control_loop():
307:     """Цикл управления ПЛК №2 (100 мс)."""
308:     while True:
309:         # Чтение данных от ПЛК №1
310:         read_from_plc1()
312:         # Логика управления (по варианту)
313:         if state['temperature'] > 70.0:
314:             state['pump_on'] = True
315:         elif state['temperature'] < 50.0:
316:             state['pump_on'] = False
318:         if state['level'] > 80.0:
319:             state['valve_open'] = True
320:         elif state['level'] < 30.0:
321:             state['valve_open'] = False
323:         # Отправка команд имитатору объекта
324:         if plant_socket:
325:             commands = {
326:                 'heater_power': state['heater_power'],
327:                 'pump_on': state['pump_on'],
328:                 'valve_open': state['valve_open']
329:             }
330:             try:
331:                 plant_socket.sendall(json.dumps(commands).encode('utf-8'))
332:                 data = plant_socket.recv(1024)
333:                 if data:
334:                     plant_state = json.loads(data.decode('utf-8'))
335:                     state['level'] = plant_state['level']
336:                     state['pressure'] = plant_state['pressure']
337:             except Exception as e:
338:                 print(f"Ошибка связи с имитатором: {e}")
340:         # Запись в Modbus-регистры ПЛК №2
341:         store.setValues(3, 0, [int(state['temperature'] * 10)])
342:         store.setValues(3, 1, [int(state['level'])])
343:         store.setValues(3, 2, [int(state['pressure'] * 100)])
344:         store.setValues(2, 0, [1 if state['pump_on'] else 0])
345:         store.setValues(2, 1, [1 if state['valve_open'] else 0])
347:         time.sleep(0.1)
350: def modbus_server():
351:     """Запуск Modbus TCP сервера ПЛК №2."""
352:     StartTcpServer(context, address=("127.0.0.1", 5021))
355: if __name__ == '__main__':
356:     # Подключение к имитатору
357:     connect_to_plant()
359:     # Запуск Modbus-сервера в отдельном потоке
360:     t_server = threading.Thread(target=modbus_server, daemon=True)
361:     t_server.start()
363:     print("ПЛК №2 запущен. Modbus TCP на порту 5021.")
365:     # Запуск цикла управления
366:     control_loop()
367: Пояснение: ПЛК №2 читает данные от ПЛК №1 (через Modbus TCP), принимает решение о включении насоса и клапана, отправляет команды имитатору объекта и выставляет свои регистры в Modbus TCP на порту 5021.
